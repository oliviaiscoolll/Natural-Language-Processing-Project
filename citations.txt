/**
* Reads a student data file and separates each line
* into name, grade level, and GPA using the "|" symbol.
* The parsed data is stored in ArrayLists.
* @param fileName name of the student file being read
*/
public void parseStudentFile(String fileName) {

// Convert file into a list of lines
ArrayList<String> lines = FileReader.toStringList(fileName);

// Temporary lists to store parsed data
ArrayList<String> nameList = new ArrayList<String>();
ArrayList<Integer> gradeList = new ArrayList<Integer>();
ArrayList<Double> gpaList = new ArrayList<Double>();

// Loop through each line of the file
for (String line : lines) {

// Find positions of the separators "|"
int firstPipe = line.indexOf("|");
int secondPipe = line.indexOf("|", firstPipe + 1);

// Extract the name, grade level, and GPA
String name = line.substring(0, firstPipe);
int grade = Integer.parseInt(line.substring(firstPipe + 1, secondPipe));
double gpa = Double.parseDouble(line.substring(secondPipe + 1));

// Store the extracted data into lists
nameList.add(name);
gradeList.add(grade);
gpaList.add(gpa);
}

// Assign parsed lists to instance variables
this.nameList = nameList;
this.gradeList = gradeList;
this.gpaList = gpaList;
}
